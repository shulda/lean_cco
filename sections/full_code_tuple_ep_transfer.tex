% Shared full unary-term code, Ramsey transfer, and separate EP obligations.
% Independent of CCO, EPPA and any specific MI/MR geometry.
\section{Full-code tuples: reconstruction, Ramsey and EP}
\label{sec:cofinal-fullcode-ep}

We first treat reconstruction and Ramsey transfer for a full tuple code.
Neither uses an expansion property or a cofinality assumption. The later
EP statements introduce an unexpanded reduct and the additional same-host
and cofinality hypotheses explicitly.

\subsection{Full term codes and generated embeddings}

Let $\mathcal H^o$ be any hereditary class of finite induced structures
in a finite many-sorted language of unary partial functions and unary/binary relations
(no nullary symbols); a hidden order, when present, is one of these binary
relations. There are finitely many visible root tags $s$. For each tag fix
a finite tuple of hidden input sorts and a quantifier-free formula
$\delta_s$ selecting the valid tuples. A root is a tagged tuple, with no
further quotient of tuples. For $D^o\in\mathcal H^o$, let $N(D^o)$ contain
\emph{all} valid root tuples.

A coordinate term first selects a coordinate of a root and then applies
a well-sorted unary term. Assume uniformly finitely many unary term normal
forms from each input sort, including the identity: every such term agrees
with its chosen normal form as a \emph{partial function}, with the same
domain and value in every member of $\mathcal H^o$. In particular this
applies to each composite of a normal form with a named unary function,
including a nowhere-defined composite when one occurs.

The \emph{full code} retains root tags and every applicable hidden unary,
binary, term-equality and order predicate. For example,
\[
\begin{aligned}
 U_\tau(x)&\quad\Longleftrightarrow\quad
                 \tau(x)\text{ is defined and }U(\tau(x)),\\
 R_{\tau,\sigma}(x,y)&\quad\Longleftrightarrow\quad
       \tau(x),\sigma(y)\text{ are defined and }R(\tau(x),\sigma(y)),\\
 E_{\tau,\sigma}(x,y)&\quad\Longleftrightarrow\quad
       \tau(x),\sigma(y)\text{ are defined and }\tau(x)=\sigma(y).
\end{aligned}
\]
Here ``applicable'' means that the term has the appropriate input sort;
its definedness at a particular root is \emph{recorded}, not assumed.
In particular $E_{\tau,\tau}(x,x)$ is the unary definedness code for
$\tau$. All same-root instances are retained, either as binary diagonals
or as unary predicates. The code is finite and unary/binary, regardless
of tuple dimension. Let $\mathcal V^o$ be the hereditary age of finite
induced subsets of these full-code $N(D^o)$, with no additional freely
interpreted visible relations. For a visible root set $A^o$ write $h(A^o)$
for the induced hidden support generated by its root coordinates.

\begin{lemma}[Full unary-term code reconstructs generated embeddings]
\label{lem:foundational-full-code-embeddings}
In the full-code setting above, the code of $A^o$ reconstructs $h(A^o)$
up to unique isomorphism over the root coordinates. For every
$D^o\in\mathcal H^o$ there is a natural bijection
\[
 \boxed{\quad
 \Emb_{\mathcal V^o}(A^o,N(D^o))
  \cong
 \Emb_{\mathcal H^o}(h(A^o),D^o).
 \quad}\tag{FCODE}\label{eq:foundational-full-code-embeddings}
\]
Neither Ramsey nor EP is assumed. Multiary hidden functions are not
covered by this binary code theorem.
\end{lemma}
\begin{proof}
The hidden support consists of the \emph{defined} coordinate-term
presentations $\tau(a)$. Definedness codes specify which presentations
occur, and the full equality codes identify exactly the repetitions,
including those from different roots. Unary and binary codes recover
all relations. If $F$ is a named unary function and $\tau(a)$ is a defined
presentation, the normal form of $F\circ\tau$ records both whether
$F(\tau(a))$ is defined and its value when defined. Thus the code also
recovers all partial-function domains and values, and hence the entire
induced support.

Given an embedding $f:A^o\hookrightarrow N(D^o)$, define
$\bar f(\tau(a))=\tau(f(a))$ on the defined presentations. Domain and
equality codes make it well-defined and injective. Relational codes and
normal-form composition make it an induced hidden embedding, reflecting
function domains as well as values. Conversely, a hidden embedding
$g:h(A^o)\hookrightarrow D^o$ sends each source root to a tuple satisfying
its quantifier-free compatibility formula. Distinct roots remain distinct:
their tags differ or an identity-coordinate value differs. The resulting
root map preserves and reflects every code. These constructions are
mutual inverses and commute with composition, because the hidden support
is generated by the root coordinates.
\end{proof}

\begin{remark}[Definedness and complete coding]
Finite normal forms alone do not record partial-function domains: already
on one point, an undefined $F$ differs from a fixed point of $F$.
The full code distinguishes them by $E_{F,F}(x,x)$. When function domains
are determined by sorts and all intended functions are total there, these
definedness codes are redundant. More generally, any code may be omitted
only after proving that it is uniformly quantifier-free definable from
the retained relations, so that the embedding sets are unchanged. This includes
replacing non-order codes by original fine visible relations in a concrete
application; the reconstruction theorem does not supply that definability.
\end{remark}

\begin{theorem}[Finite-tuple hidden Ramsey readout]
\label{thm:ramsey-tuple-readout}
In the full-code setting of \cref{lem:foundational-full-code-embeddings},
if $\mathcal H^o$ is Ramsey for embeddings, then so is $\mathcal V^o$.
No cofinal root cover, EP, visible SAP, or additional order on the visible
tuples is required.
\end{theorem}
\begin{proof}
Given $A^o,B^o\in\mathcal V^o$ and $k$, choose $D^o\in\mathcal H^o$ with
\[
                 D^o\longrightarrow(h(B^o))^{h(A^o)}_k.
\]
By \eqref{eq:foundational-full-code-embeddings}, any colouring of
$\Emb(A^o,N(D^o))$ is a colouring of $\Emb(h(A^o),D^o)$.
A monochromatic embedding $g:h(B^o)\hookrightarrow D^o$ induces an
embedding of $B^o$ into $N(D^o)$. For every $a:A^o\hookrightarrow B^o$,
the lifted map $h(a)$ lands inside $h(B^o)$, and its composite with $g$
has the common colour. Naturality of the same bijection therefore gives
the desired visible arrow; the empty input is immediate.
\end{proof}

A hidden witness may contain orphan coordinates outside $h(N(D^o))$;
they do not affect this argument. The full code, rather than an arbitrary
interpretation or a choice of one representative per tuple class, supplies
the required correspondence of \emph{all} embeddings.

\begin{lemma}[Cofinality from an ambient root cover]
\label{lem:foundational-root-cover-cofinal}
Let $P$ be an (arbitrary possibly infinite) hidden
structure with finitely many unary-term normal forms
from each root-coordinate sort. Let $\mathcal H$ be
the hereditary age of finite \emph{unary-closed}
induced substructures of $P$, and suppose every
finite $D\in\mathcal H$ embeds into $P$.
Assume each hidden element $z\in P$ is a unary-term
value $\tau(\eta(r))$ of \emph{some valid visible
root tuple} $\eta(r)\in N(P)$.
Then every finite $D\in\mathcal H$ embeds into a
finite $E\in\mathcal H$ with
\[
                         h(N(E))=E .
\tag{RC}\label{eq:foundational-root-cover}
\]
No Ramsey, expansion property, CCO or SAP hypothesis
is involved.
\end{lemma}
\begin{proof}
Embed $D$ into $P$. For each of the finitely many
elements $z\in D$ choose a valid root tuple
$r_z\in N(P)$ and an applicable unary term $\tau_z$
with $\tau_z(r_z)=z$. Let $E$ be the \emph{unary
closure in $P$} of all coordinates of all $r_z$.
The uniform finite bound on unary-term normal forms
makes $E$ finite and unary-closed; it is an induced
member of $\mathcal H$ containing the chosen copy
of $D$. Every $r_z$ is a valid tuple of elements of
$E$ since its compatibility formula is quantifier-free
and $E$ is induced. Thus $r_z\in N(E)$, and
$z\in h(N(E))$. Conversely every point of $E$ is
by construction a unary-term value of some coordinate
of one of these $r_z$, so
$E\subseteq h(N(E))$. The reverse inclusion
holds because $E$ is unary-closed. This proves
the asserted equality on the \emph{same} $E$.
\end{proof}

\subsection{Unexpanded supports and the extra obligations for EP}

For the remaining statements let $\mathcal H^o\to\mathcal H$ be a
hereditary hidden expansion, and let $\mathcal V^o\to\mathcal V$ be the
corresponding full-code visible expansion over a chosen unexpanded fine
visible age. The tuple tags and compatibility formulas are fixed in the
unexpanded hidden language. In particular $N(E)$ is determined before
choosing an expansion of $E$. The assertions below explicitly require
that forgetting the expansion from a generated support recovers its
canonical unexpanded support; this is not automatic for arbitrary
expansions introducing additional functions. No Ramsey or EP assumption
is imposed on these expansions until stated.

\begin{lemma}[Same-host lifting from the complete root code]
\label{lem:foundational-same-host-code}
Let $E\in\mathcal H$ be finite and
$B=N(E)|_{\mathcal V}$ its full \emph{unexpanded}
visible root interpretation. Assume
\[
                         h(N(E))=E,
\]
and that the \emph{unexpanded} fine-coloured
visible object $B$ determines the hidden
unexpanded generated support canonically,
up to its unique isomorphism over root-coordinate
presentations. For every allowed expanded representation $B^o$, assume
also that the reduct of $h(B^o)$ is this canonical unexpanded support,
under the same root-coordinate identification. This holds automatically
for relational hidden expansions retaining the unary functions.
Suppose allowed visible expansions
are precisely the hereditary induced \emph{full
unary-term codes} of finite hidden expanded objects
of $\mathcal H^o$, with all hidden term equality,
function, unary/binary relation and comparison
information retained. Then every allowed
expansion $B^o$ of this fixed $B$ has a hidden
realisation $E^o\in\mathcal H^o$ on the
\emph{same underlying unexpanded $E$}, and its
root code on $B$ is precisely $B^o$.

Such $E^o$ is unique as a structure on $E$
once the chosen identification of $E$ with
the unexpanded root-generated template is fixed.
\end{lemma}
\begin{proof}
Since $B^o$ belongs to the induced full-code
visible age, it has a hidden expanded
representation in some finite hidden
host $D^o$, with $B^o$ embedded into
$N(D^o)$. Apply
\cref{lem:foundational-full-code-embeddings}:
its unary-generated support $h(B^o)$
is an actual finite hidden \emph{expanded}
substructure of $D^o$, uniquely reconstructed
by its root term presentations and their complete
codes. Forgetting the expansion, the support is
canonically the unexpanded hidden template $h(B)$,
which equals $E$ by the hypotheses.
Transport the induced expanded hidden relations,
functions and order from $h(B^o)$ across this
root-coordinate identification to the
\emph{existing unexpanded set} $E$. The
result is an allowed $E^o$, by heredity of
$\mathcal H^o$, and its full visible code
is $B^o$ by construction. Every hidden point
of $E$ is a unary-term value of a visible root,
so any two expanded hidden structures on $E$
giving the same complete root code coincide on
every unary, binary, equality and comparison
instance. This proves uniqueness.
\end{proof}

\begin{theorem}[Cofinal full-code EP transfer]
\label{thm:foundational-cofinal-tuple-EP}
Assume $\mathcal H^o\to\mathcal H$ and
$\mathcal V^o\to\mathcal V$ are reasonable precompact expansions,
and that the hidden expansion has EP. Suppose the tuple interpretation
and its full unary-term, equality, relational and comparison code satisfy:
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item for every visible $A\in\mathcal V$ the unexpanded
isomorphism type of $h(A)$ is determined by the
unexpanded visible fine-coloured $A$, and the expansion
$A^o$ induces one compatible hidden expansion of $h(A)$;
\item every finite hidden $D\in\mathcal H$ embeds into
a finite $E\in\mathcal H$ such that
\[
                         h(N(E))=E;
\]
\item \emph{every} allowed visible expansion of
$N(E)$, for such a full root-generated $E$, lifts to
a hidden $\mathcal H^o$ expansion of the \emph{same
underlying unexpanded $E$}, preserving all full
term/equality/relational/comparison codes;
\item every visible embedding extends \emph{uniquely}
to the unary-generated hidden supports, sends each
valid visible root tuple to a valid root tuple,
and the tuple interpretation recovers every original
fine visible binary atom.
\end{enumerate}
Then $\mathcal V^o$ has the expansion property
relative to its unexpanded age $\mathcal V$.
This result requires neither CCO nor SAP, and
does not assert the four hypotheses for any
particular MI/MR interpretation.
\end{theorem}
\begin{proof}
Fix an allowed finite expanded visible object
$A^o\in\mathcal V^o$. Let $H_A^o$ be its
associated finite hidden expansion of the
generated support $h(A)$, as supplied by (i).
By the hidden expansion property choose
a finite unexpanded $D\in\mathcal H$ such that
\emph{every} allowed hidden expansion $D^o$
contains an embedding of $H_A^o$.
Use (ii) to embed $D$ into a finite
$E\in\mathcal H$ with $h(N(E))=E$.
Choose the unexpanded visible host
\[
                              B=N(E)|_{\mathcal V}.
\]
This host is chosen \emph{before} any
competing visible expansion is presented.

Consider an arbitrary allowed expansion $B^o$
of this fixed $B$. Hypothesis (iii) lifts its
full code to an allowed hidden expansion
$E^o$ on precisely the \emph{same} unexpanded
object $E$. Restrict $E^o$ along the fixed
embedding $D\hookrightarrow E$, obtaining
an allowed hidden expansion $D^o$.
By choice of $D$, there is a hidden expanded
embedding $u:H_A^o\hookrightarrow D^o$.
Its composite with $D^o\hookrightarrow E^o$
takes every source visible root tuple to
a valid tuple in $N(E)$ by (iv).
The complete full code ensures that the
resulting map on visible roots is an
embedding $A^o\hookrightarrow B^o$,
again by (iv). This proves the required
universal quantifier over \emph{every}
competing $B^o$, with the unexpanded host
$B$ fixed in advance.

Only the abstract four-contract interface has
been used. In particular, the hidden Ramsey
property alone, the existence of \emph{some}
hidden lift on an unrelated $E'$, or the
mere ability to lift a chosen visible
expansion would not justify the conclusion.
\end{proof}

\begin{corollary}[EP for an ambient root-covered full-code interpretation]
\label{cor:foundational-root-covered-full-code-EP}
Let $\mathcal H^o\to\mathcal H$ and
$\mathcal V^o\to\mathcal V$ be reasonable precompact expansions in
the full-code setting above. Suppose:
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item the hidden expansion $\mathcal H^o\to\mathcal H$
has EP and its finite unexpanded source is represented
by unary-closed finite substructures of one ambient
hidden $P$;
\item every hidden point of $P$ is an applicable unary
term of some valid visible root in $N(P)$, with finitely
many unary-term normal forms;
\item the complete \emph{unexpanded} visible fine code
canonically reconstructs the hidden unary-generated
support and all original visible fine atoms; moreover the reduct of
each allowed expanded generated support is that same canonical
unexpanded support over its root-coordinate presentations;
\item the allowed visible expansions are precisely
the hereditary full unary-term codes induced from
hidden $\mathcal H^o$ objects, retaining all term
equalities, unary/binary relations and hidden comparison
codes, with embeddings given by those complete codes.
\end{enumerate}
Then the visible full-code expansion $\mathcal V^o$
has the expansion property over $\mathcal V$.
No Ramsey theorem, CCO, or SAP conclusion for the
visible source is used.
\end{corollary}
\begin{proof}
Hypothesis (b) and
\cref{lem:foundational-root-cover-cofinal}
give a finite cofinal root-generated $E$ above
every finite hidden $D$, proving condition (ii)
of \cref{thm:foundational-cofinal-tuple-EP}.
Hypothesis (c) gives condition (i), and
\cref{lem:foundational-same-host-code}
with (c),(d) gives condition (iii).
The full-code embedding bijection
\cref{lem:foundational-full-code-embeddings},
together with the fine-atom faithfulness in
(c), supplies condition (iv). Apply
\cref{thm:foundational-cofinal-tuple-EP}.
\end{proof}
